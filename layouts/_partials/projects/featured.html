{{- /* The home page's featured projects: the pages its params.projects references, each entry checked and its page looked up once, for both the all-projects link and the cards. errorf does not stop the template: after an error, only the bad entry is skipped (a list that is not one counts as empty), so that no template execution error buries the error message. */}} {{- $featured := .Params.projects }} {{- if and (isset .Params "projects") (not (reflect.IsSlice $featured)) }}{{ errorf "projects: the home page's projects parameter must be a list of pages" }}{{ $featured = slice }}{{ end }} {{- /* Only regular pages are projects. The words the error uses for each other kind of page an entry may reference; a kind missing here gets the generic wording, so that it is still rejected. */}} {{- $nonProjectKindWords := dict "home" "the home page itself" "section" "a section page" "term" "a taxonomy term page" "taxonomy" "a taxonomy page" "404" "the 404 error page" }} {{- $projects := slice }} {{- range $index, $entry := $featured }} {{- $entryNumber := add $index 1 }} {{- if ne (printf "%T" $entry) "string" }}{{ errorf "projects: entry %d of the home page's projects parameter must be a page reference" $entryNumber }}{{ continue }}{{ end }} {{- if eq (strings.TrimSpace $entry) "" }}{{ errorf "projects: entry %d of the home page's projects parameter must be a non-empty page reference" $entryNumber }}{{ continue }}{{ end }} {{- if ne $entry (strings.TrimSpace $entry) }}{{ errorf "projects: the home page's projects parameter references a page with a space at its start or end, which Hugo does not trim: \"%s\"" (strings.TrimSpace $entry) }}{{ continue }}{{ end }} {{- $page := $.GetPage $entry }} {{- if not $page }}{{ errorf "projects: the home page's projects parameter references a page that cannot be found (language: %s): %s. The page may not be translated into this language, or not published: a draft, a future date (scheduled) or an expired page" $.Language.Lang $entry }}{{ continue }}{{ end }} {{- if not $page.IsPage }}{{ errorf "projects: the home page's projects parameter references %s, which is not a project: \"%s\"" (index $nonProjectKindWords $page.Kind | default "a page of another type") $entry }}{{ continue }}{{ end }} {{- if and (isset $page.Params "build") (not (reflect.IsMap $page.Params.build)) }}{{ errorf "projects: the project %s declares a custom param named build, which Hugo reserves for its build options: rename this param" $entry }}{{ continue }}{{ end }} {{- $projects = $projects | append $page }} {{- end }} {{- return $projects }}
