{{- /* The home page's featured projects: the pages its params.projects references, each entry checked and its page looked up once, for both the all-projects link and the cards. errorf does not stop the template: after an error, only the bad entry is skipped (a list that is not one counts as empty), so that no template execution error buries the error message. */}} {{- $featured := .Params.projects }} {{- if and (isset .Params "projects") (not (reflect.IsSlice $featured)) }}{{ errorf "projects: the home page's projects parameter must be a list of pages" }}{{ $featured = slice }}{{ end }} {{- /* Only regular pages are projects. The words the error uses for each other kind of page an entry may reference; a kind missing here gets the generic wording, so that it is still rejected. */}} {{- $nonProjectKindWords := dict "home" "the home page itself" "section" "a section page" "term" "a taxonomy term page" "taxonomy" "a taxonomy page" "404" "the 404 error page" }} {{- $projects := slice }} {{- range $index, $entry := $featured }} {{- $entryNumber := add $index 1 }} {{- $entryProblem := partial "text-problem.html" $entry }} {{- if eq $entryProblem "not-text" }}{{ errorf "projects: entry %d of the home page's projects parameter must be a page reference (language: %s)" $entryNumber $.Language.Lang }}{{ continue }}{{ end }} {{- if eq $entryProblem "empty" }}{{ errorf "projects: entry %d of the home page's projects parameter must be a non-empty page reference (language: %s)" $entryNumber $.Language.Lang }}{{ continue }}{{ end }} {{- if eq $entryProblem "padded" }}{{ errorf "projects: entry %d of the home page's projects parameter has a page reference with whitespace at its start or end, which Hugo does not trim (language: %s): %q" $entryNumber $.Language.Lang $entry }}{{ continue }}{{ end }} {{- $page := $.GetPage $entry }} {{- if not $page }}{{ errorf "projects: the home page's projects parameter references a page that cannot be found (language: %s): %q. The project may not exist, the page may not be translated into this language, or not published: a draft, a future (scheduled) date or an expired page" $.Language.Lang $entry }}{{ continue }}{{ end }} {{- if not $page.IsPage }}{{ errorf "projects: the home page's projects parameter references %s, which is not a project (language: %s): %q" (index $nonProjectKindWords $page.Kind | default "a page of another type") $.Language.Lang $entry }}{{ continue }}{{ end }} {{- if and (isset $page.Params "build") (not (reflect.IsMap $page.Params.build)) }}{{ errorf "projects: the project %q declares a custom parameter named build, which Hugo reserves for its build options: rename this parameter" $entry }}{{ continue }}{{ end }} {{- $projects = $projects | append $page }} {{- end }} {{- return $projects }}
