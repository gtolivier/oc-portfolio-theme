{{- /* A front matter value that must be a list: the value as it is when it is one or is not set, or else an empty list, after raising the caller's error. errorf does not stop the template: the empty list leaves the caller's part empty, so that no template execution error buries the message. */}} {{- $list := .value }} {{- if and $list (not (reflect.IsSlice $list)) }}{{ errorf .error }}{{ $list = slice }}{{ end }} {{- return $list }}
