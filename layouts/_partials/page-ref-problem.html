{{- /* What is wrong with a page reference from the front matter, before it is looked up: "not-text" (a value of another type, which site.GetPage cannot take), "empty" (only whitespace) or "padded" (whitespace at its start or end, which Hugo does not trim), or "" when nothing is. Each caller words its own error. */}} {{- $problem := "" }} {{- if ne (printf "%T" .) "string" }} {{- $problem = "not-text" }} {{- else }} {{- $trimmed := strings.TrimSpace . }} {{- if eq $trimmed "" }} {{- $problem = "empty" }} {{- else if ne . $trimmed }} {{- $problem = "padded" }} {{- end }} {{- end }} {{- return $problem }}
