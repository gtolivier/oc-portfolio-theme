{{- /* What is wrong with a front matter value that must be a text, such as a page reference or a name: "not-text" (a value of another type), "empty" (only whitespace) or "padded" (whitespace at its start or end, which Hugo does not trim from a page reference), or "" when nothing is. Each caller words its own error and decides whether "padded" is one. */}} {{- $problem := "" }} {{- if ne (printf "%T" .) "string" }} {{- $problem = "not-text" }} {{- else }} {{- $trimmed := strings.TrimSpace . }} {{- if eq $trimmed "" }} {{- $problem = "empty" }} {{- else if ne . $trimmed }} {{- $problem = "padded" }} {{- end }} {{- end }} {{- return $problem }}
