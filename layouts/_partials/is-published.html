{{- /* Whether Hugo publishes the page. Rather than decode [build] render (set on the page or through a cascade), rely on Hugo: it leaves .RelPermalink empty for every value it does not publish, except "link" (lowercase only), which keeps the address but writes no page. */}} {{- $build := .Params.build }} {{- $render := "" }} {{- if reflect.IsMap $build }}{{ $render = printf "%v" $build.render }}{{ end }} {{- return and (ne .RelPermalink "") (ne $render "link") }}
